=================================================|
Deposits DAX Measures				 |
=================================================|

Total Deposits =
CALCULATE (
    SUM ( FactDeposits[Balance] ),
    LASTNONBLANK (
        DimDate[Date],
        CALCULATE ( COUNTROWS ( FactDeposits ) )
    )
)

=================================================

CASA Deposits =
CALCULATE (
    [Total Deposits],
    DimProduct[ProductName] IN { "Current", "Savings" }
)

=================================================

CASA Ratio =
DIVIDE ( [CASA Deposits], [Total Deposits] )

=================================================

Deposits PY =
CALCULATE (
    [Total Deposits],
    SAMEPERIODLASTYEAR ( DimDate[Date] )
)

=================================================

Deposit Growth % =
DIVIDE (
    [Total Deposits] - [Deposits PY],
    [Deposits PY]
)

=================================================|
Financing DAX Measures				 |
=================================================|
Total Financing =
CALCULATE (
    SUM ( FactFinancing[Outstanding] ),
    LASTNONBLANK (
        DimDate[Date],
        CALCULATE ( COUNTROWS ( FactFinancing ) )
    )
)

=================================================
Non-Performing Financing =
CALCULATE (
    [Total Financing],
    FactFinancing[Status] = "Non-Performing"
)

=================================================
NPF Ratio =
DIVIDE (
    [Non-Performing Financing],
    [Total Financing]
)


=================================================
FDR =
DIVIDE (
    [Total Financing],
    [Total Deposits]
)

=================================================
FDR Status =
SWITCH (
    TRUE (),
    ISBLANK ( [FDR] ), "No Data",
    [FDR] > 0.9, "High",
    [FDR] >= 0.5, "Balanced",
    "Low"
)
